function normalizeText(value){
  return String(value||'')
    .normalize('NFD').replace(/[\u0300-\u036f]/g,'')
    .toLocaleLowerCase('es').replace(/\s+/g,' ').trim();
}

function normalizeUrl(raw){
  try{
    var u=new URL(raw,location.href);
    u.hash='';
    u.searchParams.delete('zenmeta');
    return u.href;
  }catch(_){return String(raw||'').trim();}
}

function esc(v){
  return String(v==null?'':v)
    .replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
    .replace(/"/g,'&quot;').replace(/'/g,'&#039;');
}

function itemById(name,id){return group(name).find(function(x){return x.id===id;})||null;}

function resolveVocabulary(name,value){
  var q=normalizeText(value);
  if(!q)return null;
  var items=group(name);
  for(var i=0;i<items.length;i++){
    var x=items[i];
    var candidates=[x.id,x.label].concat(x.aliases||[]);
    if(candidates.some(function(v){return normalizeText(v)===q;}))return x.id;
  }
  return null;
}

function validateRecord(record){
  var errors=[];
  var missing=[];

  if(!record||record.contractVersion!=='1.0.0')errors.push('Versión de contrato inválida');
  if(!record||!record.identity||!record.identity.postId)errors.push('Falta postId');
  if(!record||!record.identity||!record.identity.canonicalUrl)errors.push('Falta URL canónica');
  if(!record||!record.title)errors.push('Falta título');

  var c=record&&record.classification||{};
  if(!c.primaryPillar)missing.push('Pilar principal');
  else if(!itemById('pillars',c.primaryPillar))errors.push('Pilar principal desconocido');

  if(!c.type)missing.push('Tipo');
  else if(!itemById('types',c.type))errors.push('Tipo desconocido');

  (c.relatedPillars||[]).forEach(function(id){
    if(!itemById('pillars',id))errors.push('Pilar relacionado desconocido: '+id);
    if(id===c.primaryPillar)errors.push('El Pilar principal no puede repetirse como relacionado');
  });

  var y=record&&record.temporal&&record.temporal.documentYear;
  if(y!=null&&(Number(y)<1500||Number(y)>2200))errors.push('Año documental fuera de rango');

  var statusId=record&&record.editorial&&record.editorial.status;
  if(statusId&&!itemById('statuses',statusId))errors.push('Estado desconocido');

  (record&&record.indexing&&record.indexing.concepts||[]).forEach(function(id){
    if(!itemById('concepts',id))errors.push('Concepto desconocido: '+id);
  });

  (record&&record.indexing&&record.indexing.norms||[]).forEach(function(ref){
    if(!ref||!itemById('norms',ref.normId))errors.push('Norma desconocida: '+(ref&&ref.normId||''));
    if(ref&&ref.articles&&!Array.isArray(ref.articles))errors.push('Artículos de norma inválidos');
  });

  return {
    state:errors.length?'invalid':(missing.length?'incomplete':'complete'),
    errors:errors,
    missing:missing
  };
}

}

function loadRegistry(){
  try{
    var raw=storage().getItem(STORAGE_KEY);
    var parsed=raw?JSON.parse(raw):null;
    if(parsed&&parsed.records&&typeof parsed.records==='object'){
      state.registry=parsed;
      state.registry.migrationIssues=state.registry.migrationIssues||{};
      return {loaded:true,migrated:false};
    }
  }catch(_){}

  var migration=migrateLegacyV04();
  return {loaded:false,migrated:migration.migrated,count:migration.count};
}
