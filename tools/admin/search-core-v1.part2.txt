 null;
    const articleMatch = normalized.match(/\bart(?:iculo)?\.?\s+([0-9]+[a-z-]*)\b/);
    if (articleMatch) {
      naturalArticle = articleMatch[1];
      normalized = normalized.replace(articleMatch[0], "").replace(/\s+/g, " ").trim();
      if (!filters.article) filters.article = naturalArticle;
    }

    return {
      raw: text,
      normalized,
      phrases,
      tokens: this.normalizer.tokens(normalized),
      negatives,
      filters,
      sort: "relevance"
    };
  }
}

class InMemorySearchIndex extends SearchIndex {
  constructor() {
    super();
    this.documents = new Map();
  }

  build(documents) {
    this.documents = new Map(documents.map(d => [d.id, d]));
    return this;
  }

  get(id) {

  }

  all() {
    return [...this.documents.values()];

}

// ---- src/strategies/TitleSearchStrategy.js ----
class TitleSearchStrategy extends SearchStrategy {
  constructor({normalizer, weights}) {
    super();
    this.normalizer = normalizer;
    this.weights = weights;
  }

  search(query, context) {
    if (!query.normalized && !query.phrases.length) return [];
    const hits = [];

    for (const document of context.documents) {
      const title = this.normalizer.normalize(document.title);
      let score = 0;
      const reasons = [];

      if (query.normalized && title === query.normalized) {
        score += this.weights.titleExact;
        reasons.push({kind: "titleExact", field: "title", value: document.title});
      } else if (query.normalized && title.startsWith(query.normalized)) {
        score += this.weights.titlePrefix;
        reasons.push({kind: "titlePrefix", field: "title", value: document.title});
      }

      for (const phrase of query.phrases) {
        if (title.includes(phrase)) {
          score += this.weights.titlePhrase;
          reasons.push({kind: "titlePhrase", field: "title", value: phrase});
        }
      }

      for (const token of query.tokens) {
        if (title.split(/\s+/).includes(token)) {
          score += this.weights.titleToken;
          reasons.push({kind: "titleToken", field: "title", value: token});
        }
      }

      if (score > 0) hits.push({documentId: document.id, score, reasons});
    }
    return hits;
  }
}


// ---- src/strategies/ConceptSearchStrategy.js ----
class ConceptSearchStrategy extends SearchStrategy {
  constructor({normalizer, vocabularyResolver, weights}) {
    super();
    this.normalizer = normalizer;
    this.vocabulary = vocabularyResolver;
    this.weights = weights;
  }

  search(query, context) {
    const terms = [...query.phrases, query.normalized, ...query.tokens].filter(Boolean);
    if (!terms.length) return [];
    const hits = [];

    for (const document of context.documents) {
      let score = 0;
      const reasons = [];
      for (const conceptId of document.metadata.indexing.concepts || []) {
        for (const term of terms) {
          const match = this.vocabulary.matches("concepts", conceptId, term);
          if (!match) continue;
          const weight = match.kind === "exact" ? this.weights.conceptExact : this.weights.conceptAlias;
          score += weight;
          reasons.push({
            kind: match.kind === "exact" ? "conceptExact" : "conceptAlias",
            field: "concept",
            id: conceptId,
            value: match.item.label
          });
          break;
        }
      }
      if (score > 0) hits.push({documentId: document.id, score, reasons});
    }
    return hits;
  }
}


// ---- src/strategies/NormSearchStrategy.js ----
class NormSearchStrategy extends SearchStrategy {
  constructor({vocabularyResolver, weights}) {
    super();
    this.vocabulary = vocabularyResolver;
    this.weights = weights;
  }

  search(query, context) {
    const terms = [...query.phrases, query.normalized, ...query.tokens].filter(Boolean);
    const explicitNorm = query.filters.norm;
    const article = query.filters.article;
    const hits = [];

    for (const document of context.documents) {
      let score = 0;
      const reasons = [];

      for (const ref of document.metadata.indexing.norms || []) {
        if (explicitNorm && ref.normId === explicitNorm) {
          score += this.weights.normExact;
          reasons.push({kind: "normExact", field: "norm", id: ref.normId, value: this.vocabulary.label("norms", ref.normId)});
        }

        for (const term of terms) {
          const match = this.vocabulary.matches("norms", ref.normId, term);
          if (!match) continue;
          score += match.kind === "exact" ? this.weights.normExact : this.weights.normAlias;
          reasons.push({
            kind: match.kind === "exact" ? "normExact" : "normAlias",
            field: "norm",
            id: ref.normId,
            value: match.item.label
          });
          break;
        }

        if (article && (ref.articles || []).map(String).includes(String(article))) {
          score += this.weights.articleReference;
          reasons.push({kind: "articleReference", field: "article", value: String(article), normId: ref.normId});
        }
      }

      if (score > 0) hits.push({documentId: document.id, score, reasons});
    }
    return hits;
  }
}
