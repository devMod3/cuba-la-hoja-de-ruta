      (old.concepts||[]).forEach(function(value){
        var resolved=resolveVocabulary('concepts',value);
        if(resolved)concepts.push(resolved);
        else issues.push('Concepto sin resolver: '+value);
      });
      rec.indexing.concepts=Array.from(new Set(concepts));

      if(old.norm){
        var normId=resolveVocabulary('norms',old.norm);
        if(normId){
          rec.indexing.norms=[{
            normId:normId,
            articles:old.article?[String(old.article)]:[]
          }];
        }else{
          issues.push('Norma sin resolver: '+old.norm+(old.article?' · artículo '+old.article:''));
        }
      }

      state.registry.records[id]=rec;
      if(issues.length)state.registry.migrationIssues[id]=issues;
      result.count++;
    });

    if(result.count){
      result.migrated=true;
      persist();
    }
  }catch(_){}
  return result;
}

function parsePostId(entry){
  var raw=entry&&entry.id&&entry.id.$t||'';
  var m=raw.match(/post-(\d+)/);
  return m?m[1]:raw;
}
function alternateUrl(entry){
  var links=(entry&&entry.link)||[];
  var l=links.find(function(x){return x.rel==='alternate';});
  return l?normalizeUrl(l.href):'';
}
function feedPost(entry){
  return {
    id:parsePostId(entry),
    url:alternateUrl(entry),
    title:entry&&entry.title&&entry.title.$t||'(sin título)',
    published:entry&&entry.published&&entry.published.$t||null,
    updated:entry&&entry.updated&&entry.updated.$t||null,
    labels:(entry&&entry.category||[]).map(function(c){return c.term;}).filter(Boolean)
  };
}

async function fetchFeedPage(startIndex){
  var u=new URL('/feeds/posts/default',document.baseURI||location.href);
  u.searchParams.set('alt','json');
  u.searchParams.set('max-results',String(FEED_PAGE_SIZE));
  u.searchParams.set('start-index',String(startIndex));
  u.searchParams.set('orderby','published');
  var r=await fetch(u.href,{credentials:'same-origin',cache:'no-store'});
  if(!r.ok)throw new Error('Feed HTTP '+r.status);
  return r.json();
}

  status('Leyendo artículos del blog…','info');
  var all=[],start=1,total=null,guard=0;

  while(guard++<50){
    var data=await fetchFeedPage(start);
    var feed=data&&data.feed||{};
    var entries=(feed.entry||[]).map(feedPost).filter(function(p){return p.url&&p.id;});
    all=all.concat(entries);

    if(total===null){
      total=Number(feed['openSearch$totalResults']&&feed['openSearch$totalResults'].$t||entries.length);
    }
    if(!entries.length||all.length>=total)break;
    start+=entries.length;
  }

  var seen=new Set();
  state.posts=all.filter(function(p){
    if(seen.has(p.id))return false;
    seen.add(p.id);
    return true;
  });
  state.loaded=true;

  reconcilePosts();
  renderAll();

  var health=countHealth();
  status(
    state.posts.length+' artículos · '+
    health.complete+' completos · '+
